cmake_minimum_required(VERSION 3.15)
project(native_file_drop_tests LANGUAGES CXX)
enable_testing()
add_subdirectory(../../tool/windows installer-tools)
add_library(native_file_drop ../../windows/runner/native_file_drop.cpp)
target_compile_features(native_file_drop PUBLIC cxx_std_17)
target_include_directories(native_file_drop PUBLIC ../../windows/runner)
target_compile_definitions(native_file_drop PUBLIC UNICODE _UNICODE NOMINMAX)
target_link_libraries(native_file_drop PUBLIC user32 shell32 ole32 advapi32)
add_executable(native_file_drop_test native_file_drop_test.cpp)
target_link_libraries(native_file_drop_test PRIVATE native_file_drop)
add_executable(native_file_drop_sender WIN32 native_file_drop_sender.cpp)
target_link_libraries(native_file_drop_sender PRIVATE native_file_drop)
add_test(NAME native_file_drop COMMAND native_file_drop_test)
add_executable(startup_monitor startup_monitor.cpp)
target_compile_features(startup_monitor PRIVATE cxx_std_17)
target_compile_definitions(startup_monitor PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(startup_monitor PRIVATE user32)
add_executable(installer_monitor installer_monitor.cpp)
target_compile_features(installer_monitor PRIVATE cxx_std_17)
target_compile_definitions(installer_monitor PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(installer_monitor PRIVATE user32)
